救生員新增交辦
這套功能是在你原本網站新增一個「工作日誌」分頁，內容會取代目前紙本救生員工作日誌與水質管理日誌，例如早晚班工作事項、水質每時段紀錄、PH、餘氯、加藥、反洗、正洗、補水、事故與交接事項等。

Replit 開發補充：工作日誌系統簡化邏輯與程式碼規格
一、核心邏輯總覽

請 Replit 依照這個資料流程製作：

使用者登入
↓
取得使用者可進入的館別
↓
進入「工作日誌」分頁
↓
選擇館別 / 班別
↓
系統產生今日任務
↓
人員逐項完成
↓
系統檢查必填項目
↓
全部完成後才可送出今日回報
↓
主管審核

今日任務由 5 種資料組成：

今日任務 =
每日固定事項
+ 今日主管交辦事項
+ 今日符合星期的週期任務
+ 今日應填水質時段
+ 上一班交接事項

function getTodayTasks({ userId, venueId, shiftType, date }) {
  const user = getUserById(userId);

  // 1. 權限檢查
  if (!user.allowedVenueIds.includes(venueId)) {
    throw new Error("此帳號沒有該館別權限");
  }

  // 2. 取得今天星期
  const weekday = getWeekday(date); 
  // 回傳 mon, tue, wed, thu, fri, sat, sun

  // 3. 每日固定任務
  const dailyTasks = getDailyTasks(venueId, shiftType);

  // 4. 今日主管交辦任務
  const assignedTasks = getAssignedTasks({
    venueId,
    shiftType,
    userId,
    date
  });

  // 5. 今日週期任務
  const recurringTasks = getRecurringTasks({
    venueId,
    shiftType,
    weekday
  });

  // 6. 今日水質測量時段
  const waterTasks = generateWaterTasks({
    venueId,
    date
  });

  // 7. 上一班交接事項
  const handoverTasks = getPreviousShiftHandover({
    venueId,
    shiftType,
    date
  });

  return {
    dailyTasks,
    assignedTasks,
    recurringTasks,
    waterTasks,
    handoverTasks
  };
}
四、每日固定任務邏輯
function getDailyTasks(venueId, shiftType) {
  return dailyTaskTemplates.filter(task => {
    return (
      task.venueId === venueId &&
      task.isActive === true &&
      (task.shiftType === shiftType || task.shiftType === "all")
    );
  });
}
五、今日交辦任務邏輯
function getAssignedTasks({ venueId, shiftType, userId, date }) {
  return assignedTasks.filter(task => {
    const matchVenue = task.venueId === venueId;
    const matchDate = task.assignedDate === date;
    const matchShift = task.shiftType === shiftType || task.shiftType === "all";
    const matchUser = task.assignedToUserId === null || task.assignedToUserId === userId;

    return matchVenue && matchDate && matchShift && matchUser;
  });
}
六、每週固定任務邏輯
function getRecurringTasks({ venueId, shiftType, weekday }) {
  return recurringTaskTemplates.filter(task => {
    const matchVenue = task.venueId === venueId;
    const matchShift = task.shiftType === shiftType || task.shiftType === "all";
    const matchWeekday = task.weekdays.includes(weekday);

    return (
      task.isActive === true &&
      matchVenue &&
      matchShift &&
      matchWeekday
    );
  });
}

七、水質時段產生邏輯
1. 取得今天適用的水質規則

如果同一天有多個規則，使用 priority 最小的。

例如：

priority 1 = 暑假
priority 10 = 平常日
function getActiveWaterSchedule(venueId, date) {
  const matched = waterQualitySchedules.filter(schedule => {
    return (
      schedule.venueId === venueId &&
      schedule.isActive === true &&
      date >= schedule.startDate &&
      date <= schedule.endDate
    );
  });

  if (matched.length === 0) {
    return null;
  }

  matched.sort((a, b) => a.priority - b.priority);

  return matched[0];
}
2. 產生水質測量時段
function generateWaterTasks({ venueId, date }) {
  const schedule = getActiveWaterSchedule(venueId, date);

  if (!schedule) {
    return [];
  }

  let times = [];

  // 如果後台有自訂時間點，優先用自訂時間點
  if (schedule.customTimes && schedule.customTimes.length > 0) {
    times = schedule.customTimes;
  } else {
    times = generateTimeSlots(
      schedule.startTime,
      schedule.endTime,
      schedule.intervalMinutes
    );
  }

  return times.map(time => {
    return {
      taskType: "water",
      venueId,
      date,
      scheduledTime: time,
      title: `${time} 水質測量`,
      inputType: "water_quality_form",
      required: true,
      requirePhoto: schedule.requirePhoto,
      lateMinutes: schedule.lateMinutes,
      status: getWaterTaskStatus({ venueId, date, scheduledTime: time })
    };
  });
}
3. 時間產生工具
function generateTimeSlots(startTime, endTime, intervalMinutes) {
  const result = [];

  let current = timeToMinutes(startTime);
  const end = timeToMinutes(endTime);

  while (current <= end) {
    result.push(minutesToTime(current));
    current += intervalMinutes;
  }

  return result;
}

function timeToMinutes(time) {
  const [hour, minute] = time.split(":").map(Number);
  return hour * 60 + minute;
}

function minutesToTime(minutes) {
  const hour = Math.floor(minutes / 60);
  const minute = minutes % 60;

  return `${String(hour).padStart(2, "0")}:${String(minute).padStart(2, "0")}`;
}

九、水質異常判斷邏輯
function checkWaterAbnormal(record, standards) {
  const abnormalMessages = [];

  if (record.ph < standards.ph.min || record.ph > standards.ph.max) {
    abnormalMessages.push(`PH 異常：${record.ph}`);
  }

  if (
    record.chlorine < standards.chlorine.min ||
    record.chlorine > standards.chlorine.max
  ) {
    abnormalMessages.push(`餘氯異常：${record.chlorine}`);
  }

  if (
    record.waterTemperature < standards.waterTemperature.min ||
    record.waterTemperature > standards.waterTemperature.max
  ) {
    abnormalMessages.push(`水溫異常：${record.waterTemperature}`);
  }

  return {
    isAbnormal: abnormalMessages.length > 0,
    abnormalMessages
  };
}
儲存水質時的邏輯
function saveWaterQualityRecord(input, currentUser) {
  const standards = getWaterStandards(input.venueId);

  const checkResult = checkWaterAbnormal(input, standards);

  const status = checkResult.isAbnormal ? "abnormal" : "completed";

  const record = {
    ...input,
    measuredAt: getCurrentDateTime(),
    createdBy: currentUser.id,
    status,
    abnormalReason: checkResult.abnormalMessages.join("；")
  };

  saveToDatabase("water_quality_records", record);

  // 如果異常，自動建立異常案件
  if (checkResult.isAbnormal) {
    createAbnormalCase({
      venueId: input.venueId,
      sourceType: "water_quality",
      sourceId: record.id,
      category: "水質異常",
      level: "medium",
      description: checkResult.abnormalMessages.join("；"),
      reportedBy: currentUser.id
    });
  }

  return record;
}

十一、任務完成驗證邏輯

每一個任務儲存前，都要檢查是否符合要求。

function validateTaskCompletion(task, input) {
  const errors = [];

  if (task.required) {
    if (task.inputType === "checkbox" && input.value !== true) {
      errors.push("此項目必須勾選完成");
    }

    if (task.inputType === "yes_no" && !["yes", "no"].includes(input.value)) {
      errors.push("請選擇 YES 或 NO");
    }

    if (task.inputType === "on_off" && !["on", "off"].includes(input.value)) {
      errors.push("請選擇 ON 或 OFF");
    }

    if (task.inputType === "number" && input.value === "") {
      errors.push("請輸入數字");
    }

    if (task.inputType === "textarea" && !input.value) {
      errors.push("請填寫內容");
    }

    if (task.needPhoto && (!input.photoUrls || input.photoUrls.length === 0)) {
      errors.push("此項目必須上傳照片");
    }

    if (task.inputType === "number_photo") {
      if (input.value === "") errors.push("請輸入數字");
      if (!input.photoUrls || input.photoUrls.length === 0) errors.push("請上傳照片");
    }

    if (task.inputType === "checkbox_photo") {
      if (input.value !== true) errors.push("請勾選完成");
      if (!input.photoUrls || input.photoUrls.length === 0) errors.push("請上傳照片");
    }
  }

  return errors;
}

function validateBeforeSubmit({ userId, venueId, shiftType, date }) {
  const todayTasks = getTodayTasks({
    userId,
    venueId,
    shiftType,
    date
  });

  const missingItems = [];

  // 檢查每日固定事項
  todayTasks.dailyTasks.forEach(task => {
    if (task.required && !isTaskCompleted("daily", task.id, venueId, date, shiftType)) {
      missingItems.push(`每日固定事項：${task.title}`);
    }
  });

  // 檢查主管交辦事項
  todayTasks.assignedTasks.forEach(task => {
    if (task.required && !isTaskCompleted("assigned", task.id, venueId, date, shiftType)) {
      missingItems.push(`今日交辦事項：${task.title}`);
    }
  });

  // 檢查每週固定事項
  todayTasks.recurringTasks.forEach(task => {
    if (task.required && !isTaskCompleted("recurring", task.id, venueId, date, shiftType)) {
      missingItems.push(`每週固定事項：${task.title}`);
    }
  });

  // 檢查水質紀錄
  todayTasks.waterTasks.forEach(task => {
    if (task.required && isWaterTaskDue(task.scheduledTime) && !isWaterRecordCompleted(venueId, date, task.scheduledTime)) {
      missingItems.push(`水質紀錄：${task.scheduledTime} 尚未填寫`);
    }
  });

  // 檢查上一班交接是否確認
  todayTasks.handoverTasks.forEach(task => {
    if (!task.confirmedAt) {
      missingItems.push(`救生交接：上一班交接尚未確認`);
    }
  });

  return {
    canSubmit: missingItems.length === 0,
    missingItems
  };
}
送出今日回報
function submitDailyReport({ userId, venueId, shiftType, date }) {
  const result = validateBeforeSubmit({
    userId,
    venueId,
    shiftType,
    date
  });

  if (!result.canSubmit) {
    return {
      success: false,
      message: "尚有必填項目未完成，無法送出",
      missingItems: result.missingItems
    };
  }

  const submission = {
    venueId,
    workDate: date,
    shiftType,
    submittedBy: userId,
    submittedAt: getCurrentDateTime(),
    status: "submitted",
    reviewedBy: null,
    reviewedAt: null,
    reviewNote: null
  };

  saveToDatabase("daily_report_submissions", submission);

  return {
    success: true,
    message: "今日回報已送出",
    data: submission
  };
}

水質或交辦事項可以判斷逾時。

function isLate({ date, scheduledTime, lateMinutes }) {
  const now = new Date();
  const dueTime = new Date(`${date} ${scheduledTime}`);
  dueTime.setMinutes(dueTime.getMinutes() + lateMinutes);

  return now > dueTime;
}

水質狀態：

function getWaterTaskStatus({ venueId, date, scheduledTime }) {
  const record = findWaterRecord(venueId, date, scheduledTime);

  if (record) {
    return record.status; // completed or abnormal
  }

  const schedule = getActiveWaterSchedule(venueId, date);

  if (isLate({
    date,
    scheduledTime,
    lateMinutes: schedule.lateMinutes
  })) {
    return "late";
  }

  if (!isTimeReached(scheduledTime)) {
    return "not_started";
  }

  return "pending";
}

二十一、主管審核邏輯
function approveDailyReport(submissionId, currentUser) {
  const submission = getSubmissionById(submissionId);

  submission.status = "approved";
  submission.reviewedBy = currentUser.id;
  submission.reviewedAt = getCurrentDateTime();
  submission.reviewNote = "";

  updateDatabase("daily_report_submissions", submissionId, submission);

  return submission;
}
function returnDailyReport(submissionId, currentUser, reason) {
  if (!reason) {
    throw new Error("退回時必須填寫原因");
  }

  const submission = getSubmissionById(submissionId);

  submission.status = "returned";
  submission.reviewedBy = currentUser.id;
  submission.reviewedAt = getCurrentDateTime();
  submission.reviewNote = reason;

  updateDatabase("daily_report_submissions", submissionId, submission);

  return submission;
}

核心邏輯：

1. 使用現有登入帳號。
2. 使用者只能進入自己有權限的館別。
3. 工作日誌首頁是「今日工作頁」。
4. 今日工作頁資料來源包含：
   - 每日固定事項
   - 今日主管交辦事項
   - 今日符合星期的每週固定事項
   - 今日水質測量時段
   - 上一班救生交接事項
5. 每個任務都有 inputType，前端依 inputType 自動產生填寫元件。

10. 每個水質時段都要獨立填寫。
11. 水質填寫包含：餘氯、PH、水溫、室溫、加藥時間、加藥量、洗腳池換水、巡視更衣室、照片、備註。

14. 如果異常，自動建立異常案件。

16. 今日交辦事項依 assignedDate、venueId、shiftType、assignedToUserId 判斷是否顯示。
17. 所有必填任務完成前，不可送出今日回報。
18. 送出前執行 validateBeforeSubmit。
19. validateBeforeSubmit 需列出缺少項目，例如「水質紀錄：10:00 尚未填寫」。
20. 救生交接需支援：

   - 本班交接給下一班，可填文字、重要性、是否需主管注意、照片
